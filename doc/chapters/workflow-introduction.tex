\chapter[From a Parameter Hyper-rectangle to Finite Constraints]{From a Parameter Hyper-rectangle to Finite Constraints}
\label{sec:bernstein-motivation}

\term{pd-lmi} requires a matrix inequality to hold at every point of a continuous parameter domain. The package represents the known data and decision variables on a hyper-rectangle with cell-wise Bernstein polynomials, then converts the resulting continuous condition into finite constraints for YALMIP. An \termdef{lpv} induced-$\mathcal{L}_2$-gain problem provides the first example. A general \termdef{dpd-lmi} problem is formulated with its finite certificate construction. The supporting algebra is developed in Appendices~\ref{app:bernstein-math} and~\ref{app:certificates}, and the API contracts are presented in Chapters~\ref{sec:pdmat}--\ref{sec:pdlmi}.

\section[LPV Induced-L2-Gain Motivation]{Motivation: LPV induced-$\mathcal{L}_2$-Gain}

Consider a \termdef{lpv} plant
\begin{gather}
    \begin{aligned}
        \dot x(t) & =A(\vect{\rho}(t))x(t)+B_w(\vect{\rho}(t))w(t), \\
        z(t)       & =C_z(\vect{\rho}(t))x(t)+D_{zw}(\vect{\rho}(t))w(t).
    \end{aligned}
    \label{eq:lpv-l2-model}
\end{gather}
where $x\in\mathbb R^{n_x}$, $w\in\mathbb R^{n_w}$, $z\in\mathbb R^{n_z}$. The system matrices depend on the scheduling parameter $\vect{\rho}\in\mathbb R^\ell$, which is constrained to a hyper-rectangle $\mathcal P\subset\mathbb R^\ell$. Its derivative is constrained to another hyper-rectangle $\mathcal R\subset\mathbb R^\ell$.

Under zero initial conditions, the induced-$\mathcal{L}_2$ of such \term{lpv} system is defined as the smallest $\gamma>0$ such that
\begin{gather*}
    \lVert z\rVert_{\mathcal L_2}<\gamma\lVert w\rVert_{\mathcal L_2}
\end{gather*}

A well known sufficient condition to find such $\gamma$, is to search for a differentiable parameter dependent matrix $P(\vect{\rho})=P(\vect{\rho})^\top$ such that the following conditions hold for all $(\vect{\rho},\dot{\vect{\rho}})\in\mathcal P\times\mathcal R$\cite{scherer1996}:
\begin{gather}
    P(\vect{\rho}) \succ0,
    \label{eq:canonical-positive}\\
    \begin{bmatrix}
        \dot P(\vect{\rho},\dot{\vect{\rho}})+\operatorname{He}\!\left(P(\vect{\rho})A(\vect{\rho})\right) &
        P(\vect{\rho})B_w(\vect{\rho})                                                                             &
        C_z(\vect{\rho})^\top                                                                                        \\
        B_w(\vect{\rho})^\top P(\vect{\rho})                                                                       &
        -\gamma I_{n_w}                                                                                                    &
        D_{zw}(\vect{\rho})^\top                                                                                         \\
        C_z(\vect{\rho})                                                                                               &
        D_{zw}(\vect{\rho})                                                                                            &
        -\gamma I_{n_z}
    \end{bmatrix} \prec0.
    \label{eq:lpv-spine}
\end{gather}
where $\operatorname{He}(X)=X+X^\top$ and
\begin{gather*}
    \dot P(\vect{\rho},\dot{\vect{\rho}})
    =\sum_{s=1}^{\ell}\dot\rho_s
    \frac{\partial P}{\partial\rho_s}(\vect{\rho}).
\end{gather*}
This application is one instance of the general derivative-dependent \term{pd-lmi} below. The strict signs are analysis notation. The package uses non-strict semidefinite comparisons and leaves the numerical margin realization to the user.

\section{General Problem and Assumptions}
\label{sec:general-problem-assumptions}

Let $\vect{y}(\vect{\rho})=\begin{bmatrix}
    y_1(\vect{\rho})&\ldots&y_N(\vect{\rho})
\end{bmatrix}^\top$ be a parameter-dependent $N$-dimensional decision vector. The goal of this package is to formulate a finite certificate for the following \term{dpd-lmi} problem:
\begin{gather}
    \mathcal F(\vect{\rho},\dot{\vect{\rho}};\vect{y})
    =F_0(\vect{\rho})
    +\sum_{k=1}^{N}F_k(\vect{\rho})y_k(\vect{\rho})+\sum_{k=1}^{N}\sum_{s=1}^{\ell}
    T_{k,s}(\vect{\rho})\dot\rho_s
    \frac{\partial y_k}{\partial\rho_s}(\vect{\rho})
    \preceq0,
    \quad
    \forall(\vect{\rho},\dot{\vect{\rho}})\in\mathcal P\times\mathcal R.
    \label{eq:general-pdlmi}
\end{gather}
The package uses the following assumptions throughout the modeling and certification process.
\begin{itemize}
    \item The scheduling set and rate set are hyper-rectangles.
          \begin{gather*}
              \mathcal P=\prod_{s=1}^{\ell}[\underline\rho_s,\overline\rho_s],
              \qquad
              \mathcal R=\prod_{s=1}^{\ell}[\underline v_s,\overline v_s],
              \qquad
              \mathcal V_{\mathcal R}
              =\prod_{s=1}^{\ell}\{\underline v_s,\overline v_s\}.
          \end{gather*}
    % \item The functions $F_0$, $F_k$, and $T_{k,s}$ are known parameter-dependent matrix functions of compatible dimensions, and $\vect{y}$ is a parameter-dependent decision vector. Setting every $T_{k,s}\equiv0, \forall k,s$ removes the derivative term and recovers an ordinary derivative-free \term{pd-lmi}.
    \item The decision vector $\vect{y}$ is continuous and is represented on the chosen grid by cell-wise polynomials of the selected degree. Its values agree on shared faces, although its one-sided derivatives may differ.
    \item The known data $F_0$, $F_k$, and $T_{k,s}$ are continuous and are represented on the chosen grid by cell-wise polynomials of appropriate degree.
    \item MATLAB and YALMIP use non-strict semidefinite comparisons. A theoretical strict condition therefore requires an explicit user-selected margin, for example $-\mathcal F(\vect{\rho},\dot{\vect{\rho}};\vect{y})-\varepsilon I\succeq0$. Failure of a finite certificate leaves feasibility of \cref{eq:general-pdlmi} inconclusive.
\end{itemize}

\section[Gridding the Parameter Domain]{Gridding the Parameter Domain}
\label{sec:intro-gridding}
\index{physical grid}\index{physical node}\index{physical cell}\index{tensor grid}

For each parameter direction $s=1,\ldots,\ell$, choose an independent, possibly nonuniform axis grid
\begin{gather*}
    \mathcal G_s=\{\rho_s^{(1)},\ldots,\rho_s^{(k_s)}\},
    \qquad
    \underline\rho_s=\rho_s^{(1)}<\cdots<\rho_s^{(k_s)}=\overline\rho_s.
\end{gather*}
Their Cartesian product is
\begin{gather*}
    \mathcal G=\mathcal G_1\times\cdots\times\mathcal G_\ell,
\end{gather*}
which contains $\prod_{s=1}^{\ell}k_s$ physical nodes. The multi-index $\vect{c}=(c_1,\ldots,c_\ell)$, with $c_s\in\{1,\ldots,k_s-1\}$, defines one physical cell and its width in parameter direction $s$:
\begin{gather*}
    \mathcal H_{\vect{c}}
    =\prod_{s=1}^{\ell}
    [\rho_s^{(c_s)},\rho_s^{(c_s+1)}],
    \qquad
    h_s^{\vect{c}}
    =\rho_s^{(c_s+1)}-\rho_s^{(c_s)}>0.
\end{gather*}
Thus $\mathcal G$ has $\prod_{s=1}^{\ell} k_s$ physical nodes and determines $\prod_{s=1}^{\ell}(k_s-1)$ physical cells whose union is exactly $\mathcal P$. Gridding remains a modeling approximation as it restricts the search for $\vect{y}$ from a large $\mathcal{P}$ to a bunch of selected smaller physical cells $\mathcal{H}_{\vect{c}}$.

For example, take the following nonuniform two-dimensional grid on $[0,1]^2$:
\begin{gather*}
    \mathcal G_1=\{0,0.4,1\},
    \qquad
    \mathcal G_2=\{0,0.25,1\}.
\end{gather*}
Their tensor grid $\mathcal G=\mathcal G_1\times\mathcal G_2$ has nine physical nodes and exactly partitions $[0,1]^2$ into four physical cells. The cell indices are $(1,1)$, $(1,2)$, $(2,1)$, and $(2,2)$.

Each physical cell has its own local coordinate $\vect{\alpha}\in[0,1]^\ell$ which will be defined in the next section, and a separate finite coefficient set.

\section[Bernstein Basis on Each Cell]{Bernstein Basis on Each Cell}
\label{sec:continuous-bernstein-representation}
\index{Bernstein basis}\index{cell-wise storage}\index{local Bernstein label}

Fix a physical cell $\mathcal H_{\vect{c}}$. As different physical cells may have different side lengths, we need to first express every cell on the same domain. Define the forward affine map
\begin{gather*}
    \phi_{\vect{c}}:[0,1]^\ell\longrightarrow\mathcal H_{\vect{c}},
    \qquad
    \phi_{\vect{c},s}(\alpha_s)
    =\rho_s^{(c_s)}+h_s^{\vect{c}}\alpha_s.
\end{gather*}
Its inverse defines the local coordinate $\vect{\alpha}=\phi_{\vect{c}}^{-1}(\vect{\rho})$, where $\alpha_s=\left[\rho_s-\rho_s^{(c_s)}\right]/h_s^{\vect{c}}$, and maps every physical hyper-rectangle to the unit box $[0,1]^\ell$. Since $\vect{c}$ is arbitrary, this normalization gives a uniform coordinate domain for all cell-wise expressions.

On this common unit box, we use the normalized tensor-product Bernstein basis
\begin{gather*}
    B_{\vect{i}}^{\vect{m}}(\vect{\alpha})
    =\prod_{s=1}^{\ell}
    B_{i_s}^{m_s}(\alpha_s)
    =\prod_{s=1}^{\ell}\binom{m_s}{i_s}(1-\alpha_s)^{m_s-i_s}\alpha_s^{i_s},
    \qquad
    \vect{i}\in\prod_{s=1}^{\ell}\{0,\ldots,m_s\},
\end{gather*}
where the vector $\vect{m}=(m_1,\ldots,m_\ell)$ specifies the modeled degree in each parameter direction.

For any cell-wise quantity $X$, write its pullback as $X^{(\vect{c})}=X\circ\phi_{\vect{c}}$. A cell-wise polynomial is then represented by
\begin{gather}
    X^{(\vect{c})}(\vect{\alpha})
    =\sum_{\vect{i}\in\prod_{s=1}^{\ell}\{0,\ldots,m_s\}}
    B_{\vect{i}}^{\vect{m}}(\vect{\alpha})X^{(\vect{c})}[\vect{i}].
    \label{eq:intro-cell-bernstein}
\end{gather}
Each cell stores $\prod_{s=1}^{\ell}(m_s+1)$ coefficients $X^{(\vect{c})}[\vect{i}]$. These local labels identify coefficient positions inside the cell and are separate from the physical sample points. The unequal components of $\vect{m}$ allow different polynomial degrees in different parameter directions.

The polynomial expressed in the Bernstein basis in \cref{eq:intro-cell-bernstein} is called a \textbf{\textit{Bernstein polynomial}}. On a boundary of the unit box, only the coefficients on the corresponding face are active. Continuity on the shared face of two adjacent cells therefore reduces to equality of the corresponding coefficients. The Bernstein representation also supports exact cell-wise differentiation, which is required by the derivative terms in a \term{dpd-lmi}.

To make the cell-wise storage concrete, return to the two-dimensional grid in \cref{fig:intro-two-dimensional-grid} and consider
\begin{gather*}
    a(\rho_1,\rho_2)=1+\rho_1^3+\rho_2+\rho_1\rho_2.
\end{gather*}
Choose the anisotropic degree $\vect{m}=(3,1)$, written as \code{Degree=[3 1]} in MATLAB. Every physical cell then has eight local labels $\vect{i}=(i_1,i_2)\in\{0,1,2,3\}\times\{0,1\}$. The following input constructs one known \code{pdmat} on all four cells and inspects the coefficients stored for cell $[1,1]$.

\begin{lstlisting}[style=dpmatlab]
grid = {[0 0.4 1], [0 0.25 1]};
a = pdmat(grid, @(rho1, rho2) ...
    1 + rho1.^3 + rho2 + rho1.*rho2, ...
    Degree=[3 1]);

labelCoefficientMap = [a.lbls(), ...
    cell2mat(a.LocalValues{1}{1}).']
\end{lstlisting}

\begin{lstlisting}[style=dpoutput]
labelCoefficientMap = 8×3 double
         0         0    1.0000
         0    1.0000    1.2500
    1.0000         0    1.0000
    1.0000    1.0000    1.2833
    2.0000         0    1.0000
    2.0000    1.0000    1.3167
    3.0000         0    1.0640
    3.0000    1.0000    1.4140
\end{lstlisting}

In \cref{fig:intro-two-dimensional-grid}, cell $[1,1]$ is $\mathcal H_{(1,1)}=[0,0.4]\times[0,0.25]$. Its forward map is $\phi_{(1,1)}(\vect{\alpha})=(0.4\alpha_1,0.25\alpha_2)$, so the pullback of $a$ to the unit box is
\begin{gather*}
    a^{(1,1)}(\alpha_1,\alpha_2)
    =1+0.064\alpha_1^3+0.25\alpha_2+0.1\alpha_1\alpha_2.
\end{gather*}
Substituting the eight stored coefficients into \cref{eq:intro-cell-bernstein} gives the Bernstein form
\begin{gather*}
    \begin{aligned}
        a^{(1,1)}(\alpha_1,\alpha_2)={}&
        B_0^3(\alpha_1)B_0^1(\alpha_2)
        +\frac{5}{4}B_0^3(\alpha_1)B_1^1(\alpha_2) \\
        &+B_1^3(\alpha_1)B_0^1(\alpha_2)
        +\frac{77}{60}B_1^3(\alpha_1)B_1^1(\alpha_2) \\
        &+B_2^3(\alpha_1)B_0^1(\alpha_2)
        +\frac{79}{60}B_2^3(\alpha_1)B_1^1(\alpha_2) \\
        &+\frac{133}{125}B_3^3(\alpha_1)B_0^1(\alpha_2)
        +\frac{707}{500}B_3^3(\alpha_1)B_1^1(\alpha_2).
    \end{aligned}
\end{gather*}
Each row of \code{labelCoefficientMap} gives one term in this expansion. Its first two columns contain the local label $(i_1,i_2)$, and its third column contains the corresponding coefficient from \code{a.LocalValues{1}{1}}. Thus in the cell $\mathcal{H}_{(1,1)}$, \code{pdmat} stores the coefficients for labels $(0,0),(0,1),(1,0),(1,1),(2,0),(2,1),(3,0),(3,1)$ in that order. The two nested indices select the physical cell, while the eight leaf columns select local Bernstein labels inside that cell. These eight values represent the polynomial over the complete cell and leave the physical grid unchanged. \Cref{sec:bernstein-tensor-grids,sec:bernstein-higher-degree,sec:bernstein-degree-elevation,sec:bernstein-product-math} give the complete traversal, higher-degree, elevation, and multiplication rules.
